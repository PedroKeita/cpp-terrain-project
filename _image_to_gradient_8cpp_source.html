80Eigen::MatrixXd ImageToGradient::computeMagnitude(const Eigen::MatrixXd& dx,
81 const Eigen::MatrixXd& dy) {
82 // Validate input dimensions
83 if (dx.rows() != dy.rows() || dx.cols() != dy.cols()) {
84 throw std::invalid_argument("ImageToGradient::computeMagnitude: "
85 "dx and dy matrices must have same dimensions");
86 }
87
88 int rows = dx.rows();
89 int cols = dx.cols();
90
91 Eigen::MatrixXd magnitude(rows, cols);
92
93 // Compute magnitude for each pixel: sqrt(dx² + dy²)
94 for (int y = 0; y < rows; ++y) {
95 for (int x = 0; x < cols; ++x) {
96 double dxVal = dx(y, x);
97 double dyVal = dy(y, x);
98 magnitude(y, x) = std::sqrt(dxVal * dxVal + dyVal * dyVal);
