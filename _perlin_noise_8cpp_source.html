112double PerlinNoise::grad(int hash, double x, double y, double z) {
113 int h = hash & 15;
114 double u = h < 8 ? x : y;
115 double v = h < 4 ? y : (h == 12 || h == 14 ? x : z);
116 return ((h & 1) == 0 ? u : -u) + ((h & 2) == 0 ? v : -v);
117}

133double PerlinNoise::noise(double x, double y, double z) {
134
135 // Locates grid cells
136 int X = (int)floor(x) & 255;
137 int Y = (int)floor(y) & 255;
138 int Z = (int)floor(z) & 255;
139
140 // Internal distances
141 x -= floor(x);
142 y -= floor(y);
143 z -= floor(z);
144
145 // Smoothing curves
146 double u = fade(x);
147 double v = fade(y);
148 double w = fade(z);
149
150 // Hashes of the 8 cell corners
151 int A = p[X] + Y;
152 int AA = p[A] + Z;
153 int AB = p[A + 1] + Z;
154 int B = p[X + 1] + Y;
155 int BA = p[B] + Z;
156 int BB = p[B + 1] + Z;
157
158 // Complete trilinear interpolation
159 return lerp(w, lerp(v, lerp(u, grad(p[AA], x, y, z),
160 grad(p[BA], x - 1, y, z)),
161 lerp(u, grad(p[AB], x, y - 1, z),
162 grad(p[BB], x - 1, y - 1, z))),
163 lerp(v, lerp(u, grad(p[AA + 1], x, y, z - 1),
164 grad(p[BA + 1], x - 1, y, z - 1)),
165 lerp(u, grad(p[AB + 1], x, y - 1, z - 1),
166 grad(p[BB + 1], x - 1, y - 1, z - 1))));
167}

183double PerlinNoise::octaveNoise(double x, double y, int octaves, double persistence) {
184 double total = 0.0;
185 double frequency = 1.0;
186 double amplitude = 1.0;
187 double maxValue = 0.0;
188
189 for (int i = 0; i < octaves; i++) {
190 total += noise(x * frequency, y * frequency, 0.0) * amplitude;
191 maxValue += amplitude;
192 amplitude *= persistence;
193 frequency *= 2.0;
194 }
195
196 return total / maxValue;
197}
